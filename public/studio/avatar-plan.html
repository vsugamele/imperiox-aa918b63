heygen-adapter.js · estrutura baseJS
const HG = axios.create({
  baseURL: 'https://api.heygen.com',
  headers: { 'x-api-key': process.env.HEYGEN_API_KEY }
});

// Gerar vídeo com Photo Avatar + script narrado
async function gerarAvatarVideo(narration, avatarId, voiceId) {
  const { data } = await HG.post('/v2/video/generate', {
    video_inputs: [{
      character: {
        type:      'avatar',
        avatar_id: avatarId,           // ID do seu Photo Avatar
        avatar_style: 'normal'
      },
      voice: {
        type:     'text',
        input_text: narration,         // narração do storyboard
        voice_id: voiceId,             // voz HeyGen ou ElevenLabs clone
      },
    }],
    dimension: { width: 1080, height: 1920 }, // 9:16
    test: false,
  });
  return data.data.video_id;
}

// Polling do status
async function pollVideo(videoId) {
  for (let i = 0; i < 30; i++) {
    await sleep(5000);
    const { data } = await HG.get(`/v1/video_status.get?video_id=${videoId}`);
    if (data.data.status === 'completed') return data.data.video_url;
    if (data.data.status === 'failed')    throw new Error(data.data.error);
  }
}

// Lipsync — animar qualquer vídeo com áudio novo
async function lipsync(videoUrl, audioUrl) {
  const { data } = await HG.post('/v2/video/translate', {
    video_url: videoUrl,
    audio_url: audioUrl,   // URL pública do MP3 do ElevenLabs
    title:     'lipsync_cena',
  });
  return data.data.video_id;
}

Insight de custo

O HeyGen Avatar III ($1.00/min) é muito mais barato que o Seedance para cenas de talking head — porque um clip de 30s do HeyGen custa $0.50 vs $3.60 de 30s no Seedance. A lógica se inverte: use Seedance para cenas de ambiente/movimento de câmera, use HeyGen para quando tem um avatar falando.
